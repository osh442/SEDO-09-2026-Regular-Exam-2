pipeline {
    agent any

    triggers {
        pollSCM('* * * * *')
    }

    environment {
        DOTNET_CLI_TELEMETRY_OPTOUT = '1'
        DOTNET_NOLOGO = '1'
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }
        stage('Restore') {
            steps {
                bat 'dotnet restore Homies.sln'
            }
        }
        stage('Build') {
            steps {
                bat 'dotnet build Homies.sln --configuration Release --no-restore'
            }
        }
        stage('Test') {
            steps {
                bat 'dotnet test Homies.sln --configuration Release --no-build --verbosity normal'
            }
        }
    }
}
